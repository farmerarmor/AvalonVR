cmake_minimum_required(VERSION 3.24)
project(AvalonVR VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
option(AVALON_GPU_VALIDATION "Build standalone GPU validation tools" OFF)

add_library(minhook STATIC vendor/minhook/src/buffer.c vendor/minhook/src/hook.c vendor/minhook/src/trampoline.c vendor/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC vendor/minhook/include)
add_library(avalon_foveation STATIC src/Foveation.cpp)
target_include_directories(avalon_foveation PUBLIC src PRIVATE vendor/nvapi)
target_compile_definitions(avalon_foveation PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(avalon_foveation PUBLIC d3d11 d3dcompiler "${CMAKE_SOURCE_DIR}/vendor/nvapi/amd64/nvapi64.lib")

add_library(AvalonVR SHARED
 src/Bootstrap.cpp src/StereoCamera.cpp src/OpenXRBridge.cpp src/UIOverlay.cpp
 src/NativeDLSS.cpp src/DLSSDiagnostics.cpp src/VRStatus.cpp src/NativeAimUI.cpp
 src/ShaderHunter.cpp src/ShadowCulling.cpp src/ShadowResolution.cpp
 src/CharacterPoolDiagnostics.cpp src/CharacterLifecycle.cpp src/GameLoading.cpp
 src/NativeVrsPass.cpp src/QuadExposure.cpp src/BufferUploads.cpp src/QuadFog.cpp)
target_include_directories(AvalonVR PRIVATE vendor/openxr/include)
target_compile_definitions(AvalonVR PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(AvalonVR PRIVATE minhook avalon_foveation d3d11 dxgi d3dcompiler "${CMAKE_SOURCE_DIR}/vendor/openxr/lib/openxr_loader.lib" advapi32 ole32 shell32 user32 gdi32)
add_executable(AvalonVRLauncher src/Launcher.cpp)
target_link_libraries(AvalonVRLauncher PRIVATE d3d11 dxgi)
target_compile_definitions(AvalonVRLauncher PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
add_library(AvalonVRCharacterPools SHARED src/KandraPoolBootstrap.cpp)
target_link_libraries(AvalonVRCharacterPools PRIVATE minhook)
target_compile_definitions(AvalonVRCharacterPools PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(AvalonVR AvalonVRLauncher AvalonVRCharacterPools PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

enable_testing()
foreach(test IN ITEMS VRMath RadialShadow CharacterPool CharacterLifecycle LoadingGate FoveationMask QuadViewMath QuadSampling QuadFog)
 add_executable(${test}Tests tests/${test}Tests.cpp)
 target_include_directories(${test}Tests PRIVATE src)
 target_compile_options(${test}Tests PRIVATE /UNDEBUG)
 add_test(NAME ${test} COMMAND ${test}Tests)
endforeach()
add_executable(AvalonVRHunterControlsTest tools/HunterControlsValidation.cpp)
target_include_directories(AvalonVRHunterControlsTest PRIVATE src)
add_test(NAME HunterControls COMMAND AvalonVRHunterControlsTest)

if(AVALON_GPU_VALIDATION)
 foreach(kind IN ITEMS Hunter MarkedEffect GroundFog)
  add_executable(AvalonVR${kind}GpuTest tools/${kind}GpuValidation.cpp src/ShaderHunter.cpp)
  target_include_directories(AvalonVR${kind}GpuTest PRIVATE src)
  target_compile_definitions(AvalonVR${kind}GpuTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(AvalonVR${kind}GpuTest PRIVATE minhook avalon_foveation d3d11 dxgi d3dcompiler user32)
 endforeach()
 foreach(kind IN ITEMS BufferUpload QuadView Overlay)
  add_executable(AvalonVR${kind}GpuTest tools/${kind}GpuTest.cpp)
  target_include_directories(AvalonVR${kind}GpuTest PRIVATE src)
  target_compile_definitions(AvalonVR${kind}GpuTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(AvalonVR${kind}GpuTest PRIVATE d3d11 dxgi d3dcompiler gdi32 user32)
 endforeach()
 add_executable(AvalonVRFoveationGpuTest tools/FoveationGpuTest.cpp)
 target_compile_definitions(AvalonVRFoveationGpuTest PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
 target_link_libraries(AvalonVRFoveationGpuTest PRIVATE avalon_foveation dxgi)
endif()
